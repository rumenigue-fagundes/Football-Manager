package "futebol" {

    package "clube" {
        package "domain" {
            class Clube
        }
        package "application" {
            class ClubeService
        }
    }

    package "jogador" {
        package "domain" {
            abstract class Jogador
            class Goleiro
            class JogadorLinha
        }
        package "application" {
            class JogadorService
        }
    }

    package "campeonato" {
        package "domain" {
            class Campeonato
        }
        package "application" {
            class CampeonatoService
        }
    }

    package "partida" {
        package "domain" {
            class Partida
            class Escalacao
            class Participacao
        }
        package "application" {
            class PartidaService
        }
    }

    package "estatisticas" {
        package "domain" {
            class EstatisticasClube
            class EstatisticasJogador
        }
        package "application" {
            class EstatisticasService
        }
    }

    package "shared" {
        package "application" {
            interface "Repositorio<T>" as Repositorio {
                + salvar(objeto: T): void
                + buscarPorId(id: int): T
                + listarTodos(): List<T>
                + remover(id: int): void
            }
        }
        package "infrastructure" {
            class "RepositorioArquivo<T>" as RepositorioArquivo
            class "RepositorioMemoria<T>" as RepositorioMemoria
        }
    }

    package "presentation" {

        package "factory" {
            interface FabricaApresentacao {
                + criarMenu(): Menu
                + criarTelaClubes(): TelaClubes
                + criarTelaJogadores(): TelaJogadores
                + criarTelaCampeonatos(): TelaCampeonatos
                + criarTelaPartidas(): TelaPartidas
            }

            class FabricaTextual
            class FabricaGUI

            class ConfiguracaoSistema {
                {static} - usarGUI: boolean
                {static} + obterFabrica(): FabricaApresentacao
            }

            FabricaTextual ..|> FabricaApresentacao
            FabricaGUI ..|> FabricaApresentacao
            ConfiguracaoSistema ..> FabricaApresentacao
        }

        package "produtos" {
            interface Menu {
                + exibir(): void
            }
            interface TelaClubes {
                + exibir(): void
            }
            interface TelaJogadores {
                + exibir(): void
            }
            interface TelaCampeonatos {
                + exibir(): void
            }
            interface TelaPartidas {
                + exibir(): void
            }
        }

        package "textual" {
            class MenuTextual
            class TelaClubesTextual
            class TelaJogadoresTextual
            class TelaCampeonatosTextual
            class TelaPartidasTextual

            MenuTextual ..|> Menu
            TelaClubesTextual ..|> TelaClubes
            TelaJogadoresTextual ..|> TelaJogadores
            TelaCampeonatosTextual ..|> TelaCampeonatos
            TelaPartidasTextual ..|> TelaPartidas
        }

        package "gui" {
            class MenuGUI
            class TelaClubesGUI
            class TelaJogadoresGUI
            class TelaCampeonatosGUI
            class TelaPartidasGUI

            MenuGUI ..|> Menu
            TelaClubesGUI ..|> TelaClubes
            TelaJogadoresGUI ..|> TelaJogadores
            TelaCampeonatosGUI ..|> TelaCampeonatos
            TelaPartidasGUI ..|> TelaPartidas
        }
    }
}
